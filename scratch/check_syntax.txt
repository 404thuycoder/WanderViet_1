                        id: el.id,
                        name: tags.name || `Địa điểm ${cat}`,
                        lat: el.lat,
                        lng: el.lon,
                        address: addr,
                        type: cat,
                        distanceText: d < 1000 ? `${Math.round(d)}m` : `${(d/1000).toFixed(1)}km`,
                        travelTime: d < 500 ? '5 phút đi bộ' : d < 1000 ? '12 phút đi bộ' : '5 phút taxi',
                        rating: (4 + Math.random() * 0.9).toFixed(1),
                        reviews: Math.floor(Math.random() * 500) + 50,
                        image: `https://source.unsplash.com/400x300/?${cat},city,${tags.name?encodeURIComponent(tags.name):''}`
                    };
                });
                renderNearbyUI(currentNearbyData);
            } catch (err) {
                console.error('Map search error:', err);
                grid.style.opacity = '1';
            }
        }

            nearbyMarkers.forEach(m => map.removeLayer(m));
            nearbyMarkers = [];
        }
