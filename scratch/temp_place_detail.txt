        async function doPay() {
            const name = document.getElementById('c-name').value;
            if(!name) { alert("Vui lòng nhập tên!"); return; }
            
            const btn = document.getElementById('btn-pay-main');
            btn.disabled = true;
            btn.innerText = 'ĐANG XỬ LÝ...';

            document.getElementById('p-form').style.display = 'none';
            document.getElementById('p-status').style.display = 'flex';
            document.getElementById('st-loading').style.display = 'block';
            document.getElementById('st-success').style.display = 'none';
            document.querySelector('.modal-content').scrollTop = 0;
            
            try {
                let pMethod = currentM;
                if (currentM === 'counter') pMethod = 'contact';
                if (currentM === 'qr') pMethod = 'transfer';

                const r = await fetch('/api/bookings', {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json', 'Authorization': `Bearer ${localStorage.getItem('wander_token')}` },
                    body: JSON.stringify({ placeId: pid, customerName: name, customerPhone: document.getElementById('c-phone').value, useDate: document.getElementById('c-date').value, paymentMethod: pMethod })
                });
                const j = await r.json();
                if(j.success) {
                    setTimeout(() => {
                        document.getElementById('st-loading').style.display = 'none';
                        document.getElementById('st-success').style.display = 'block';
                        document.getElementById('b-id').innerText = '#WV-' + j.data.bookingId.split('-').pop().toUpperCase();
                        
                        // Zenith Effects
                        const chime = new Audio('https://assets.mixkit.co/active_storage/sfx/2013/2013-preview.mp3');
                        chime.volume = 0.5;
                        chime.play().catch(e=>console.log('Audio wait'));
                        
                        document.getElementById('success-card').classList.add('success-pulse');
                        setTimeout(() => document.getElementById('lux-shimmer').classList.add('shimmer-active'), 500);
                        
                        const container = document.getElementById('confetti-container');
                        const colors = ['#6366f1', '#10b981', '#f59e0b', '#ec4899', '#ffffff'];
                        
                        // Ambient Floating
                        for(let i=0; i<8; i++) {
                            const ap = document.createElement('div');
                            ap.className = 'ambient-p';
                            ap.style.width = ap.style.height = (2+Math.random()*4) + 'px';
                            ap.style.left = (Math.random()*100) + '%';
                            ap.style.bottom = '-10px';
                            ap.style.animation = `floatingP ${3+Math.random()*5}s linear infinite`;
                            ap.style.animationDelay = (Math.random()*2) + 's';
                            document.getElementById('success-card').appendChild(ap);
                        }

                        for(let i=0; i<32; i++) {
                            const p = document.createElement('div');
                            p.className = 'p-confetti';
                            const angle = Math.random() * Math.PI * 2;
                            const dist = 60 + Math.random() * 120;
                            p.style.setProperty('--tx', Math.cos(angle) * dist + 'px');
                            p.style.setProperty('--ty', Math.sin(angle) * dist + 'px');
                            p.style.backgroundColor = colors[Math.floor(Math.random()*colors.length)];
                            p.style.left = '50%'; p.style.top = '50%';
                            p.style.animationDelay = (Math.random() * 0.3) + 's';
                            container.appendChild(p);
                        }
                    }, 800);
                } else { alert(j.message); document.getElementById('p-status').style.display = 'none'; document.getElementById('p-form').style.display = 'block'; }
            } catch(e) { document.getElementById('p-status').style.display = 'none'; document.getElementById('p-form').style.display = 'block'; }
        }

        // ── Render Reviews ─────────────────────────────────────────────
        function renderReviews(reviews, p) {
            const list = document.getElementById('reviews-list');
            const avgEl = document.getElementById('review-avg');
            const totalEl = document.getElementById('review-total');
            const posEl = document.getElementById('review-positive');
            if (!list) return;

            const total = reviews.length;
            if (total === 0) {
                list.innerHTML = '<p style="color:var(--text-dim);text-align:center;padding:32px 0">Chưa có đánh giá nào. Hãy là người đầu tiên!</p>';
                if (avgEl) avgEl.innerText = parseFloat(p.ratingAvg || 5.0).toFixed(1);
                if (totalEl) totalEl.innerText = '0 đánh giá';
                if (posEl) posEl.innerText = '0%';
                renderStarDistribution([], p.ratingAvg || 5.0);
                return;
            }

            const avgRating = reviews.reduce((s,r) => s + (r.rating||5), 0) / total;
            const positiveCount = reviews.filter(r => (r.rating||5) >= 4).length;
            if (avgEl) avgEl.innerText = avgRating.toFixed(1);
            if (totalEl) totalEl.innerText = `${total} đánh giá`;
            if (posEl) posEl.innerText = Math.round(positiveCount/total*100) + '%';
            document.getElementById('p-rate').innerText = avgRating.toFixed(1);
            document.getElementById('p-review-count').innerText = `(${total} đánh giá)`;

            // Render star distribution bars
            renderStarDistribution(reviews, avgRating);

            list.innerHTML = reviews.map(r => {
                const stars = '⭐'.repeat(Math.min(5, Math.max(1, r.rating || 5)));
                const initials = (r.userName || 'K').charAt(0).toUpperCase();
                const date = r.createdAt ? new Date(r.createdAt).toLocaleDateString('vi-VN') : '';
                const sentiment = r.rating >= 4 ? 'positive' : r.rating === 3 ? 'neutral' : 'negative';
                return `
                <div class="review-item">
                    <div class="review-header">
                        <div class="review-avatar">${initials}</div>
                        <div class="review-user-info">
                            <div class="review-user-name">
                                ${r.userName || 'Khách hàng'}
                                ${r.verified ? '<span class="review-verified">✓ Đã xác minh</span>' : ''}
                                <span class="review-sentiment ${sentiment}" style="margin-left:auto">${sentiment==='positive'?'Tích cực':sentiment==='neutral'?'Trung lập':'Tiêu cực'}</span>
                            </div>
                            <div class="review-date">${date}</div>
                        </div>
                    </div>
                    <div class="review-rating"><span>${stars}</span></div>
                    <div class="review-text">${r.text || ''}</div>
                    ${r.images && r.images.length ? `<div class="review-images">${r.images.map(img=>`<img src="${img}" onerror="this.onerror=null; this.src='https://images.unsplash.com/photo-1528127269322-539801943592?w=300&q=80'" alt="review">`).join('')}</div>` : ''}
                    <div class="review-actions">
                        <span class="review-action" onclick="this.innerHTML='👍 '+((parseInt(this.innerHTML.match(/\\d+/)?.[0]||0))+1)+' Hữu ích'">👍 ${r.helpfulCount||0} Hữu ích</span>
                    </div>
                </div>`;
            }).join('');
        }

        function renderStarDistribution(reviews, avgRating) {
            // Count reviews per star level
            const counts = {5:0, 4:0, 3:0, 2:0, 1:0};
            if (reviews.length === 0) {
                // Simulate distribution based on avgRating for demo
                const simTotal = 20;
                counts[5] = Math.round(simTotal * (avgRating >= 4.5 ? 0.7 : 0.4));
                counts[4] = Math.round(simTotal * 0.2);
                counts[3] = Math.round(simTotal * 0.06);
                counts[2] = Math.round(simTotal * 0.02);
                counts[1] = simTotal - counts[5] - counts[4] - counts[3] - counts[2];
            } else {
                reviews.forEach(r => { const s = Math.min(5, Math.max(1, r.rating || 5)); counts[s]++; });
            }

            const total = reviews.length || 20;
            [1,2,3,4,5].forEach(s => {
                const fill = document.getElementById(`dist-${s}`);
                const num = document.getElementById(`dist-${s}-n`);
                if (fill) {
                    const pct = Math.round(counts[s]/total*100);
                    fill.style.width = pct + '%';
                    fill.classList.add('animated');
                }
                if (num) num.innerText = counts[s];
            });

            // Update the score label
            const scoreLabel = avgRating >= 4.5 ? 'Xuất sắc' : avgRating >= 4 ? 'Rất tốt' : avgRating >= 3 ? 'Tốt' : 'Bình thường';
            const labelEl = document.querySelector('.review-stats div[style*="font-size:12px"]');
            if (labelEl) labelEl.innerText = scoreLabel;
        }

            function renderItems(filter = 'all') {
                const filtered = filter === 'all' ? items : items.filter(it => (it.category || '').toLowerCase() === filter);

        function getEmoji(type) {
            if (type.includes('restaurant') || type.includes('food')) return '🍽️';
            if (type.includes('hotel') || type.includes('tourism')) return '🏨';
            if (type.includes('cafe')) return '☕';
            if (type.includes('attraction') || type.includes('viewpoint')) return '🎯';
            return '✨';
        }

        function getDistance(lat1, lon1, lat2, lon2) {
            const R = 6371e3; // meters
            const φ1 = lat1 * Math.PI/180;
            const φ2 = lat2 * Math.PI/180;
            const Δφ = (lat2-lat1) * Math.PI/180;
            const Δλ = (lon2-lon1) * Math.PI/180;
            const a = Math.sin(Δφ/2) * Math.sin(Δφ/2) +
                      Math.cos(φ1) * Math.cos(φ2) *
                      Math.sin(Δλ/2) * Math.sin(Δλ/2);
            const c = 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1-a));
            return R * c;
        }

        function focusOnMarker(lat, lng, name) {
            if (!map) return;
            map.flyTo([lat, lng], 17, { duration: 1.5 });
            nearbyMarkers.forEach(m => {
                if (Math.abs(m.getLatLng().lat - lat) < 0.0001 && Math.abs(m.getLatLng().lng - lng) < 0.0001) {
                    m.openPopup();
                }
            });
        }

        async function likePost(postId) {

        async function submitFavorite() {
            if (!localStorage.getItem('wander_token')) return window.WanderUI.openAuthModal();

                if (json.success) {
                    // Update count inline — no need to reload
                    const countEl = document.getElementById('p-fav-count');
                    if (countEl) countEl.innerText = json.favoritesCount;
                    // Update button visual
                    if (favBtn) {
                        const icon = favBtn.querySelector('span');
                        if (json.isFavorited) {
                            if (icon) icon.textContent = '💖';
                            favBtn.style.background = 'linear-gradient(135deg, rgba(239, 68, 68, 0.4), rgba(220, 38, 38, 0.2))';
                            favBtn.style.borderColor = 'rgba(239, 68, 68, 0.6)';

                            if (icon) icon.textContent = '❤️';
                            favBtn.style.background = 'linear-gradient(135deg, rgba(239, 68, 68, 0.15), rgba(220, 38, 38, 0.05))';
                            favBtn.style.borderColor = 'rgba(239, 68, 68, 0.3)';
                            favBtn.style.boxShadow = '0 4px 15px rgba(239,68,68,0.1)';
                        }
                    }
                    window.WanderUI.showToast(json.isFavorited ? '💖 Đã thêm vào Yêu thích!' : '💔 Đã bỏ yêu thích', json.isFavorited ? 'success' : 'info');

        async function submitBusinessFeedback() {
            const msgInput = document.getElementById('fb-modal-msg');
            const msg = msgInput.value.trim();
            if (!msg) return window.WanderUI.showToast('Vui lòng nhập nội dung.', 'error');
            
            const btn = document.getElementById('fb-submit-btn');
            btn.innerHTML = '<span class="loader-inline"></span> Đang gửi...';
            btn.disabled = true;

            try {
                const res = await fetch(`/api/feedback`, {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json', 'x-auth-token': localStorage.getItem('wander_token') },
                    body: JSON.stringify({
                        message: `[Từ dịch vụ: ${document.getElementById('p-title').innerText}] ` + msg,
                        targetRole: 'business',
                        businessId: window.currentPlace ? window.currentPlace.ownerId : null,
                        image: window.currentPlace ? window.currentPlace.image : null // Add image
                    })
                });
                const json = await res.json();
                if (json.success) {
                    window.WanderUI.showToast('Đã gửi yêu cầu tới doanh nghiệp!', 'success');
                    closeBusinessFeedbackModal();
                    msgInput.value = '';
                } else window.WanderUI.showToast(json.message, 'error');
            } catch(e) { window.WanderUI.showToast('Lỗi hệ thống', 'error'); }
            
            btn.innerHTML = 'Gửi yêu cầu';
            btn.disabled = false;
        }

        function selectPkg(type, el) {
            selectedPkg = type;
            document.querySelectorAll('.pkg-option').forEach(p => p.classList.remove('active'));
            el.classList.add('active');

            document.getElementById('p-price').innerText = newPrice.toLocaleString('vi-VN') + ' ₫';
        }

        function initPricing(basePrice) {
            basePriceValue = basePrice || 0;
            const fmt = (n) => n.toLocaleString('vi-VN') + ' ₫';
            const standard = basePriceValue;
            const premium  = Math.round(basePriceValue * 1.5);
            const luxury   = Math.round(basePriceValue * 2.2);

            document.getElementById('pkg-standard-price').innerText = fmt(standard);
            document.getElementById('pkg-premium-price').innerText  = fmt(premium);
            document.getElementById('pkg-luxury-price').innerText   = fmt(luxury);

            // Show discount if price > 0
            if (basePrice > 0) {
                const discountPct = 20;
                const originalPrice = Math.round(basePrice * (100 / (100 - discountPct)));
                const savings = originalPrice - basePrice;
                document.getElementById('p-price-original').innerText = fmt(originalPrice);
                document.getElementById('p-price-original').style.display = 'block';
                document.getElementById('price-discount-badge').style.display = 'inline-block';
                document.getElementById('price-discount-badge').innerText = `-${discountPct}%`;
                document.getElementById('price-savings').innerText = `✅ Tiết kiệm ${fmt(savings)} so với đặt trực tiếp`;
                document.getElementById('price-savings').style.display = 'block';
            }
        }

        // Performance: Lazy Loading with Intersection Observer
        document.addEventListener('DOMContentLoaded', () => {
            // Lazy load images
            const lazyImages = document.querySelectorAll('img[data-src]');
            const imageObserver = new IntersectionObserver((entries, observer) => {
                entries.forEach(entry => {
                    if (entry.isIntersecting) {
                        const img = entry.target;
                        img.src = img.dataset.src;
                        img.classList.add('loaded');
                        observer.unobserve(img);
                    }
                });
            }, { rootMargin: '50px 0px', threshold: 0.01 });

            lazyImages.forEach(img => imageObserver.observe(img));

            // Lazy load sections
            const lazySections = document.querySelectorAll('.lazy-load');
            const sectionObserver = new IntersectionObserver((entries, observer) => {
                entries.forEach(entry => {
                    if (entry.isIntersecting) {
                        entry.target.classList.add('visible');
                        observer.unobserve(entry.target);
                    }
                });
            }, { rootMargin: '100px 0px', threshold: 0.1 });

            lazySections.forEach(section => sectionObserver.observe(section));
        });

        // Add loading="lazy" to all images for native lazy loading
        document.querySelectorAll('img').forEach(img => {
            if (!img.hasAttribute('loading')) {
                img.setAttribute('loading', 'lazy');
            }
        });
