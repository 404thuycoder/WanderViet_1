        function clearMarkers() {
            nearbyMarkers.forEach(m => map.removeLayer(m));
            nearbyMarkers = [];
        }

        function renderNearbyUI(data) {
            const grid = document.getElementById('nearby-grid');
            
            // Add markers
            data.forEach(n => {
                const iconHtml = `<div style="background:#fff; width:32px; height:32px; border-radius:50%; border:2px solid var(--primary); display:flex; align-items:center; justify-content:center; font-size:16px; box-shadow:0 4px 15px rgba(0,0,0,0.4);">${getEmoji(n.type)}</div>`;
                const icon = L.divIcon({ className: 'nearby-pin', html: iconHtml, iconSize: [32, 32], iconAnchor: [16, 32] });
                const m = L.marker([n.lat, n.lng], { icon }).addTo(map)
                    .bindPopup(`<b>${n.name}</b><br><span style="font-size:12px;color:#666;">${n.address}</span>`, { closeButton: false });
                nearbyMarkers.push(m);
            });

            grid.innerHTML = data.slice(0, 15).map(n => `
                <div class="nearby-card" style="background:rgba(255,255,255,0.02); border:1px solid var(--glass-border); border-radius:20px; overflow:hidden; transition:0.3s; cursor:pointer;" onclick="focusOnMarker(${n.lat}, ${n.lng}, '${n.name}')">

                        <div style="position:absolute; bottom:12px; left:12px; background:rgba(0,0,0,0.6); backdrop-filter:blur(10px); padding:4px 8px; border-radius:8px; font-size:11px; color:#fff;">${getEmoji(n.type)} ${n.type}</div>

        function getEmoji(type) {
            if (type.includes('restaurant') || type.includes('food')) return '🍽️';
            if (type.includes('hotel') || type.includes('tourism')) return '🏨';
            if (type.includes('cafe')) return '☕';
            if (type.includes('attraction') || type.includes('viewpoint')) return '🎯';
            return '✨';
        }

        function getDistance(lat1, lon1, lat2, lon2) {
            const R = 6371e3; // meters
            const φ1 = lat1 * Math.PI/180;
            const φ2 = lat2 * Math.PI/180;
            const Δφ = (lat2-lat1) * Math.PI/180;
