        // Read More Toggle
        function toggleReadMore() {
            const descContainer = document.getElementById('p-desc-container');
            const btn = document.getElementById('read-more-btn');
            
            if (descContainer.style.maxHeight === '300px') {
                descContainer.style.maxHeight = 'none';
                descContainer.style.overflow = 'visible';
                btn.innerText = 'Thu gọn ▲';
            } else {
                descContainer.style.maxHeight = '300px';
                descContainer.style.overflow = 'hidden';
                btn.innerText = 'Đọc thêm ▼';
            }
        }

        function handleCommunityImageUpload(event) {

        async function submitCommunityPost() {
            if (!localStorage.getItem('wander_token')) return window.WanderUI.openAuthModal();
            
            const text = document.getElementById('community-post-input').value.trim();
            if (!text && !communityImageFile) return window.WanderUI.showToast('Vui lòng nhập nội dung hoặc thêm ảnh', 'error');
            
            try {
                const formData = new FormData();
                formData.append('placeId', pid);
                formData.append('content', text);
                if (communityImageFile) formData.append('image', communityImageFile);
                
                const res = await fetch('/api/social/posts', {
                    method: 'POST',
                    headers: { 'x-auth-token': localStorage.getItem('wander_token') },
                    body: formData
                });
                
                const json = await res.json();
                if (json.success) {
                    window.WanderUI.showToast('Đã đăng bài thành công!', 'success');
                    document.getElementById('community-post-input').value = '';
                    communityImageFile = null;
                    loadCommunityPosts();
                } else {
                    window.WanderUI.showToast(json.message || 'Lỗi đăng bài', 'error');
                }
            } catch(e) {
                window.WanderUI.showToast('Lỗi kết nối', 'error');
            }
        }

        async function likePost(postId) {
            if (!localStorage.getItem('wander_token')) return window.WanderUI.openAuthModal();
            try {
                const res = await fetch(`/api/social/posts/${postId}/like`, {
                    method: 'POST',
                    headers: { 'x-auth-token': localStorage.getItem('wander_token') }
                });
                const json = await res.json();
                if (json.success) {
                    loadCommunityPosts();
                }
            } catch(e) {
                console.error('Like post error:', e);
            }
        }

        async function submitFavorite() {
            if (!localStorage.getItem('wander_token')) return window.WanderUI.openAuthModal();
            const favBtn = event ? event.currentTarget : document.querySelector('[onclick="submitFavorite()"]');
            if (favBtn) { favBtn.disabled = true; favBtn.style.opacity = '0.6'; }
            try {
                const res = await fetch(`/api/places/${pid}/favorite`, {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json', 'x-auth-token': localStorage.getItem('wander_token') },
                    body: JSON.stringify({})
                });
                const json = await res.json();
                if (json.success) {
                    // Update count inline — no need to reload
                    const countEl = document.getElementById('p-fav-count');
                    if (countEl) countEl.innerText = json.favoritesCount;
                    // Update button visual
                    if (favBtn) {
                        const icon = favBtn.querySelector('span');
                        if (json.isFavorited) {
                            if (icon) icon.textContent = '💖';
                            favBtn.style.background = 'linear-gradient(135deg, rgba(239, 68, 68, 0.4), rgba(220, 38, 38, 0.2))';
                            favBtn.style.borderColor = 'rgba(239, 68, 68, 0.6)';
                            favBtn.style.boxShadow = '0 4px 20px rgba(239,68,68,0.35)';
                        } else {
                            if (icon) icon.textContent = '❤️';
                            favBtn.style.background = 'linear-gradient(135deg, rgba(239, 68, 68, 0.15), rgba(220, 38, 38, 0.05))';
                            favBtn.style.borderColor = 'rgba(239, 68, 68, 0.3)';
                            favBtn.style.boxShadow = '0 4px 15px rgba(239,68,68,0.1)';
                        }
                    }
                    window.WanderUI.showToast(json.isFavorited ? '💖 Đã thêm vào Yêu thích!' : '💔 Đã bỏ yêu thích', json.isFavorited ? 'success' : 'info');
                } else {
                    window.WanderUI.showToast(json.message || 'Lỗi cập nhật yêu thích', 'error');
                }
            } catch(e) {
                window.WanderUI.showToast('Lỗi kết nối máy chủ', 'error');
            } finally {
                if (favBtn) { favBtn.disabled = false; favBtn.style.opacity = '1'; }
            }
        }

        function openBusinessFeedbackModal() {
            if (!localStorage.getItem('wander_token')) return window.WanderUI.openAuthModal();
            if (!window.currentPlace) return;
            document.getElementById('fb-modal-title').innerText = window.currentPlace.name;
            const fbImg = document.getElementById('fb-modal-img');
            fbImg.onerror = function() { this.src = 'https://images.unsplash.com/photo-1528127269322-539801943592?w=500&q=80'; };
            fbImg.src = window.currentPlace.image;
            document.getElementById('biz-feedback-modal').style.display = 'flex';
        }

        function closeBusinessFeedbackModal() {
            document.getElementById('biz-feedback-modal').style.display = 'none';
        }

        async function submitBusinessFeedback() {
            const msgInput = document.getElementById('fb-modal-msg');
            const msg = msgInput.value.trim();
            if (!msg) return window.WanderUI.showToast('Vui lòng nhập nội dung.', 'error');
            
            const btn = document.getElementById('fb-submit-btn');
            btn.innerHTML = '<span class="loader-inline"></span> Đang gửi...';
            btn.disabled = true;

            try {
                const res = await fetch(`/api/feedback`, {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json', 'x-auth-token': localStorage.getItem('wander_token') },
                    body: JSON.stringify({
                        message: `[Từ dịch vụ: ${document.getElementById('p-title').innerText}] ` + msg,
                        targetRole: 'business',
                        businessId: window.currentPlace ? window.currentPlace.ownerId : null,
                        image: window.currentPlace ? window.currentPlace.image : null // Add image
                    })
                });
                const json = await res.json();
                if (json.success) {
                    window.WanderUI.showToast('Đã gửi yêu cầu tới doanh nghiệp!', 'success');
                    closeBusinessFeedbackModal();
                    msgInput.value = '';
                } else window.WanderUI.showToast(json.message, 'error');
            } catch(e) { window.WanderUI.showToast('Lỗi hệ thống', 'error'); }
            
            btn.innerHTML = 'Gửi yêu cầu';
            btn.disabled = false;
        }

        let selectedPkg = 'standard';
        let basePriceValue = 0;

        function selectPkg(type, el) {
            selectedPkg = type;
            document.querySelectorAll('.pkg-option').forEach(p => p.classList.remove('active'));
            el.classList.add('active');
            // Update main price display based on package
            const multiplier = type === 'premium' ? 1.5 : type === 'luxury' ? 2.2 : 1;
            const newPrice = Math.round(basePriceValue * multiplier);
            document.getElementById('p-price').innerText = newPrice.toLocaleString('vi-VN') + ' ₫';
        }

        function initPricing(basePrice) {
            basePriceValue = basePrice || 0;
            const fmt = (n) => n.toLocaleString('vi-VN') + ' ₫';
            const standard = basePriceValue;
            const premium  = Math.round(basePriceValue * 1.5);
            const luxury   = Math.round(basePriceValue * 2.2);

            document.getElementById('pkg-standard-price').innerText = fmt(standard);
            document.getElementById('pkg-premium-price').innerText  = fmt(premium);
            document.getElementById('pkg-luxury-price').innerText   = fmt(luxury);

            // Show discount if price > 0
            if (basePrice > 0) {
                const discountPct = 20;
                const originalPrice = Math.round(basePrice * (100 / (100 - discountPct)));
                const savings = originalPrice - basePrice;
                document.getElementById('p-price-original').innerText = fmt(originalPrice);
                document.getElementById('p-price-original').style.display = 'block';
                document.getElementById('price-discount-badge').style.display = 'inline-block';
                document.getElementById('price-discount-badge').innerText = `-${discountPct}%`;
                document.getElementById('price-savings').innerText = `✅ Tiết kiệm ${fmt(savings)} so với đặt trực tiếp`;
                document.getElementById('price-savings').style.display = 'block';
            }
        }

        let spInterval = null;
        function startSocialProof() {
            const toast = document.getElementById('sp-toast');
            if (!toast) return;
            function show() {
                const name = spNames[Math.floor(Math.random() * spNames.length)];
                const loc  = spLocations[Math.floor(Math.random() * spLocations.length)];
                const action = spActions[Math.floor(Math.random() * spActions.length)];
                document.getElementById('sp-toast-icon').textContent = action.icon;
                document.getElementById('sp-toast-msg').innerHTML  = action.msg(name, loc);
                document.getElementById('sp-toast-sub').textContent = action.sub;
                toast.classList.add('show');
                setTimeout(() => toast.classList.remove('show'), 4500);
            }
            setTimeout(show, 3000);
            spInterval = setInterval(show, 12000);
        }

        function startViewerCounter() {
            const el = document.getElementById('viewers-count');
            if (!el) return;
            let base = 8 + Math.floor(Math.random() * 18);
            el.textContent = base + ' người';
            setInterval(() => {
                const delta = Math.floor(Math.random() * 3) - 1;
                base = Math.max(3, Math.min(40, base + delta));
                el.textContent = base + ' người';
            }, 8000);
        }
