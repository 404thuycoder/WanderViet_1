            const Δλ = (lon2-lon1) * Math.PI/180;
            const a = Math.sin(Δφ/2) * Math.sin(Δφ/2) +
                      Math.cos(φ1) * Math.cos(φ2) *
                      Math.sin(Δλ/2) * Math.sin(Δλ/2);
            const c = 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1-a));
            return R * c;
        }

        // FAQ Functions
        function toggleFAQ(index) {
            const answer = document.getElementById(`faq-answer-${index}`);
            const icon = answer.previousElementSibling.querySelector('.faq-icon');
            if (answer.style.display === 'none') {
                answer.style.display = 'block';
                icon.style.transform = 'rotate(180deg)';
            } else {
                answer.style.display = 'none';
                icon.style.transform = 'rotate(0deg)';
            }
        }

        async function voteFAQ(faqId) {
            try {
                const res = await fetch(`/api/places/${pid}/faqs/${faqId}/vote`, {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json', 'x-auth-token': localStorage.getItem('wander_token') }
                });
                const json = await res.json();
                if (json.success) {
                    window.WanderUI.showToast('Cảm ơn phản hồi của bạn!', 'success');
                }
            } catch(e) {
                console.error('FAQ vote error:', e);
            }
        }
