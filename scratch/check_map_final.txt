
                        const d = getDistance(window.currentPlaceCoords.lat, window.currentPlaceCoords.lng, el.lat, el.lon);
                        const tags = el.tags || {};
                        const cat = tags.amenity || tags.tourism || 'other';
                        const addr = tags["addr:full"] || tags["addr:street"] || tags["addr:place"] || "Vị trí trên bản đồ";
                        
                        discoveredPlaces.set(el.id, {
                            id: el.id,
                            name: tags.name || `Địa điểm ${cat}`,
                            lat: el.lat,
                            lng: el.lon,
                            address: addr,
                            type: cat,
                            website: tags.website || tags["contact:website"] || null,
                            phone: tags.phone || tags["contact:phone"] || null,
                            distanceText: d < 1000 ? `${Math.round(d)}m` : `${(d/1000).toFixed(1)}km`,
                            travelTime: d < 500 ? '5 phút đi bộ' : d < 1000 ? '12 phút đi bộ' : '5 phút taxi',
                            rating: (4.2 + Math.random() * 0.7).toFixed(1),
                            reviews: Math.floor(Math.random() * 800) + 100,
                            image: `https://source.unsplash.com/400x300/?${cat},${tags.name?encodeURIComponent(tags.name):'city'}`
                        });
                    });
                }

                renderNearbyUI();
            } catch (err) {
                console.error('Map search error:', err);
                grid.style.opacity = '1';
            }
        }

        function getEmoji(type) {
            type = type.toLowerCase();
            if (type.includes('restaurant') || type.includes('food')) return '🍽️';
            if (type.includes('hotel') || type.includes('tourism') || type.includes('accommodation')) return '🏨';
            if (type.includes('cafe') || type.includes('coffee')) return '☕';
            if (type.includes('attraction') || type.includes('viewpoint') || type.includes('museum')) return '🎯';
            if (type.includes('bank') || type.includes('atm')) return '💳';
            if (type.includes('pharmacy') || type.includes('hospital')) return '🏥';
            return '✨';
        }

        function focusOnMarker(lat, lng, id) {
            map.flyTo([lat, lng], 17, { duration: 1.2 });
            const m = discoveredMarkers.get(id);
            if (m) m.openPopup();
        }

        function getDistance(lat1, lon1, lat2, lon2) {
            const R = 6371e3;
            const φ1 = lat1 * Math.PI/180;
            const φ2 = lat2 * Math.PI/180;
            const Δφ = (lat2-lat1) * Math.PI/180;
            const Δλ = (lon2-lon1) * Math.PI/180;
            const a = Math.sin(Δφ/2) * Math.sin(Δφ/2) + Math.cos(φ1) * Math.cos(φ2) * Math.sin(Δλ/2) * Math.sin(Δλ/2);
            return R * 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1-a));
        }

        function initMap(p) {
            if (map) { map.remove(); map = null; }
            discoveredMarkers.clear();
            discoveredPlaces.clear();
            
            const lat = (p.gpsCoordinates && p.gpsCoordinates.lat) || p.lat || 21.0285;
            const lng = (p.gpsCoordinates && p.gpsCoordinates.lng) || p.lng || 105.8542;

            map = L.map('map-root', { zoomControl: false }).setView([lat, lng], 16);
            
            L.tileLayer('https://{s}.basemaps.cartocdn.com/rastertiles/voyager/{z}/{x}/{y}{r}.png', {
                attribution: '&copy; WanderViệt &copy; OpenStreetMap'
            }).addTo(map);

            L.control.zoom({ position: 'bottomright' }).addTo(map);

            const mainIcon = L.divIcon({
                className: 'main-pin',
                html: `<div style="background:var(--primary); width:32px; height:32px; border-radius:50%; border:3px solid #fff; box-shadow:0 0 15px var(--primary); display:flex; align-items:center; justify-content:center; color:#fff; font-size:18px;">📍</div>`,
                iconSize: [32, 32],
                iconAnchor: [16, 32]
            });

            mainMarker = L.marker([lat, lng], { icon: mainIcon }).addTo(map)
                .bindPopup(`<b style="color:#000;">${p.name}</b><br><span style="color:#666; font-size:12px;">Vị trí chính xác</span>`, { closeButton: false })
                .openPopup();
            
            window.currentPlaceCoords = { lat, lng };

            // Dynamic Behavior: Search when map moves/zooms (Google Maps Style)
            map.on('moveend', () => {
                if (activeFilters.size > 0) refreshMapData();
            });
        }

        function toggleNearbyFilter(type) {
            const btn = document.querySelector(`.nearby-filter[data-type="${type}"]`);
            if (activeFilters.has(type)) {
                activeFilters.delete(type);
                if (btn) btn.classList.remove('active');
            } else {
                activeFilters.add(type);
                if (btn) btn.classList.add('active');
            }
            refreshMapData();
        }
